    newGraphics = array.new<pivotGraphic>()

    for [index, coord] in pivotPointsArray

// Determine current leg direction
leg(int size) =>
    var int l = 0
    if high[size] > ta.highest(size)
        l := BEARISH_LEG
    else if low[size] < ta.lowest(size)
        l := BULLISH_LEG
    l

startOfNewLeg(int l)     => ta.change(l) != 0
startOfBearishLeg(int l) => ta.change(l) == -1
startOfBullishLeg(int l) => ta.change(l) == +1

// Store an order block from the pivot range
storeOrderBlock(pivot p_ivot, bool internal, int bias) =>
    if (not internal and showSwingOrderBlocksInput) or (internal and showInternalOrderBlocksInput)
        array<float> a_rray = na
        int parsedIndex = na

        if bias == BEARISH
            a_rray      := parsedHighs.slice(p_ivot.barIndex, bar_index)
            parsedIndex := p_ivot.barIndex + a_rray.indexof(a_rray.max())
        else
            a_rray      := parsedLows.slice(p_ivot.barIndex, bar_index)
            parsedIndex := p_ivot.barIndex + a_rray.indexof(a_rray.min())

        orderBlock ob              = orderBlock.new(parsedHighs.get(parsedIndex), parsedLows.get(parsedIndex), times.get(parsedIndex), bias)
        array<orderBlock> obArray  = internal ? internalOrderBlocks : swingOrderBlocks

        if obArray.size() >= 100
            obArray.pop()
        obArray.unshift(ob)

// Remove mitigated order blocks
deleteOrderBlocks(bool internal) =>
    array<orderBlock> obArray = internal ? internalOrderBlocks : swingOrderBlocks
    for [i, ob] in obArray
        bool crossed = false
        if bearishMitigationSrc > ob.barHigh and ob.bias == BEARISH
            crossed := true
        else if bullishMitigationSrc < ob.barLow and ob.bias == BULLISH
            crossed := true
        if crossed
            obArray.remove(i)

// Draw order blocks as boxes
drawOrderBlocks(bool internal) =>
    array<orderBlock> obArray = internal ? internalOrderBlocks : swingOrderBlocks
    int obSize = obArray.size()
    if obSize > 0
        int maxOB              = internal ? internalOrderBlocksSizeInput : swingOrderBlocksSizeInput
        array<orderBlock> parsed = obArray.slice(0, math.min(maxOB, obSize))
        array<box> boxes       = internal ? internalOrderBlocksBoxes : swingOrderBlocksBoxes

        for [i, ob] in parsed
            color obColor =
                 internal
                 ? (ob.bias == BEARISH ? internalBearishOrderBlockColor : internalBullishOrderBlockColor)
                 : (ob.bias == BEARISH ? swingBearishOrderBlockColor    : swingBullishOrderBlockColor)

            // Height Scale: compress top/bottom toward the block's own midpoint (right edge/x-values untouched)
            float midSMC = (ob.barHigh + ob.barLow) / 2
            float scaledHighSMC = midSMC + (ob.barHigh - midSMC) * boxHeightScale
            float scaledLowSMC  = midSMC - (midSMC - ob.barLow) * boxHeightScale

            box b = boxes.get(i)
            b.set_top_left_point(    chart.point.new(ob.barTime,    na, scaledHighSMC))
            b.set_bottom_right_point(chart.point.new(last_bar_time, na, scaledLowSMC))
            b.set_border_color(internal ? na : obColor)
            b.set_bgcolor(obColor)

// Track pivot points and trigger order block storage on structure breaks
getCurrentStructure(int size, bool internal) =>
    int  currentLeg = leg(size)
    bool newPivot   = startOfNewLeg(currentLeg)
    bool pivotLow   = startOfBullishLeg(currentLeg)
    bool pivotHigh  = startOfBearishLeg(currentLeg)

    if newPivot
        if pivotLow
            pivot p = internal ? internalLow : swingLow
            p.lastLevel    := p.currentLevel
            p.currentLevel := low[size]
            p.crossed      := false
            p.barTime      := time[size]
            p.barIndex     := bar_index[size]
        else
            pivot p = internal ? internalHigh : swingHigh
            p.lastLevel    := p.currentLevel
            p.currentLevel := high[size]
            p.crossed      := false
            p.barTime      := time[size]
            p.barIndex     := bar_index[size]

// Detect structure breaks and store order blocks accordingly
displayStructure(bool internal) =>
    pivot  ph = internal ? internalHigh : swingHigh
    pivot  pl = internal ? internalLow  : swingLow
    trend  t  = internal ? internalTrend : swingTrend

    // Bullish break of structure
    extraBull = internal ? internalHigh.currentLevel != swingHigh.currentLevel : true
    if ta.crossover(close, ph.currentLevel) and not ph.crossed and extraBull
        ph.crossed := true
        t.bias     := BULLISH
        if (internal and showInternalOrderBlocksInput) or (not internal and showSwingOrderBlocksInput)
            storeOrderBlock(ph, internal, BULLISH)

    // Bearish break of structure
    extraBear = internal ? internalLow.currentLevel != swingLow.currentLevel : true
    if ta.crossunder(close, pl.currentLevel) and not pl.crossed and extraBear
        pl.crossed := true
        t.bias     := BEARISH
        if (internal and showInternalOrderBlocksInput) or (not internal and showSwingOrderBlocksInput)
            storeOrderBlock(pl, internal, BEARISH)

var float SDA__ = 0.0 // Supply Distal  Aggressive
var float SPA__ = 0.0 // Supply Proximal Aggressive


if SCRO__ <= ATR__ * 0.5
    SPA__ := low[bar_index - bar_index_Supply_OB__]
    SDA__ := SmaxHigh__
    SDA__
else if SCRO__ > ATR__ * 0.5 and SCRO__ <= ATR__
    SPA__ := Candle_Body__ >= 0 ? close[bar_index - bar_index_Supply_OB__] : open[bar_index - bar_index_Supply_OB__]
    SDA__ := SmaxHigh__
    SDA__
else if SCRO__ > ATR__
    if high[bar_index - bar_index_Supply_OB__ + 1] > low[bar_index - bar_index_Supply_OB__]
        if high[bar_index - bar_index_Supply_OB__ + 1] > high[bar_index - bar_index_Supply_OB__]
            SPA__ := high[bar_index - bar_index_Supply_OB__ + 1]
            SDA__ := high[bar_index - bar_index_Supply_OB__]
            SDA__
        else if high[bar_index - bar_index_Supply_OB__ + 1] <= high[bar_index - bar_index_Supply_OB__]
            if Candle_Body__[bar_index - bar_index_Supply_OB__ + 1] > 0 or Candle_Body__[bar_index - bar_index_Supply_OB__ + 1] == 0
                if SCRa1__ > ATR__
                    SPA__ := SminClose__
                    SDA__ := SmaxHigh__
                    SDA__
                else if SCRa1__ < ATR__
                    SPA__ := SminOpen__
                    SDA__ := SmaxHigh__
                    SDA__
            else if Candle_Body__[bar_index - bar_index_Supply_OB__ + 1] < 0
                if SCRa1__ > ATR__
                    SPA__ := SminOpen__
                    SDA__ := SmaxHigh__
                    SDA__
                else if SCRa1__ < ATR__
                    SPA__ := SmaxHigh__
                    SDA__ := high[bar_index - bar_index_Supply_OB__ + 1]
                    SDA__
    else if not(high[bar_index - bar_index_Supply_OB__ + 1] > low[bar_index - bar_index_Supply_OB__])
        if SmaxHigh__ - SminOpen__ > SmaxHigh__ - SminClose__ and SmaxHigh__ - SminOpen__ <= ATR__ and SmaxHigh__ - SminOpen__ >= ATR__ * 0.5
            SPA__ := SminOpen__
            SDA__ := SmaxHigh__
            SDA__
        else if SmaxHigh__ - SminOpen__ < SmaxHigh__ - SminClose__ and SmaxHigh__ - SminClose__ <= ATR__ and SmaxHigh__ - SminClose__ >= ATR__ * 0.5
            SPA__ := SminClose__
            SDA__ := SmaxHigh__
            SDA__
else
    SPA__ := SminLow__
    SDA__ := SmaxHigh__
    SDA__


// ─── SUPPLY DEFENSIVE SHRINK ─────────────────────────────────────────────────
var float SProximal__ = 0.0
var float SDistal__ = 0.0
if RefineType__ == 'Defensive'
    float sratio__ = SDA__ - SPA__ >= ATR__ * 3 ? 0.7 : SDA__ - SPA__ >= ATR__ * 2 ? 0.6 : SDA__ - SPA__ >= ATR__ * 1.6 ? 0.5 : SDA__ - SPA__ > ATR__ ? 0.25 : 0.0
    SDistal__ := SDA__
    SProximal__ := sratio__ == 0.0 ? SPA__ : SPA__ + (SDA__ - SPA__) * sratio__
    SProximal__
else
    SDistal__ := SDA__
    SProximal__ := SPA__
    SProximal__

var float YSd12__ = 0.0 // distal  price (top)

    XSp1__ := bar_index_Supply_OB__

    XSp2__ := bar_index_Supply_OB__ + 1
    XSp2__

    supBox__ := box.new(XSp1__, math.max(YSd12__, YSp12__), XSp2__, math.min(YSd12__, YSp12__), bgcolor = color.new(#FF1744, 88), border_color = color.new(#FF1744, 100), border_width = 0, text = '', text_size = size.small, text_color = color.white, text_halign = text.align_right)
    SupCounter__ := 1
    SupCheck__ := true
    SupCheck__


if (high < YSp12__ or XSd1__ == YSp12__[1]) and SupCheck__ == true and not na(SLineDistal__)
    SupCounter__ := SupCounter__ + 1
    line.set_x2(SLineDistal__, bar_index + 1)
    line.set_x2(SLineProximal__, bar_index + 1)
    line.set_x2(SLine50per__, bar_index + 1)
    if not na(supBox__)
        box.set_right(supBox__, bar_index + 1)


if high > YSd12__ and high[1] < YSd12__
    SupCheck__ := false
    SupCheck__

// NEW indicator: VPVR : Fr3d0's Volume Profile Visible Range

LINE_COLOR_VPVR = color.rgb(255, 97, 255)

POC_LINE_COLOR_VPVR = color.rgb(255, 97, 255)

VALUE_AREA_LINE_COLOR_VPVR = color.rgb(255, 97, 255, 34)

BUY_COLOR_VPVR = color.new(color.green, 50)

SELL_COLOR_VPVR = color.new(color.red, 50)

MAX_BARS_VPVR = 365



topBottomLineColor_VPVR = input.color(LINE_COLOR_VPVR, "Top/bottom Line Color")

pocLineColor_VPVR = input.color(POC_LINE_COLOR_VPVR, "Point of Control Line Color")

valueAreaLineColor_VPVR = input.color(VALUE_AREA_LINE_COLOR_VPVR, "Value Area Line Color")

buyBarColor_VPVR = input.color(BUY_COLOR_VPVR, "Buy Bar Color")

sellBarColor_VPVR = input.color(SELL_COLOR_VPVR, "Sell Bar Color")

numOfBars_VPVR = input.int(50, 'Number of bars', minval=21, maxval=MAX_BARS_VPVR)

distLastCandle_VPVR = input.int(2, 'Distance from last candle', minval=1, maxval=MAX_BARS_VPVR)

numOfProfiles_VPVR = input.int(3, 'Number of profiles', minval=1, maxval=10)

numOfHists_VPVR = input.int(50, 'Number of hists', minval=20, maxval=100)



// Clean up drawings every tick

MAX_BOXES_LIMIT_VPVR = 500

MAX_BOXES_VPVR = numOfProfiles_VPVR * numOfHists_VPVR

expectedBoxes_VPVR = MAX_BOXES_VPVR * 2



var buyBars_VPVR = array.new_box(MAX_BOXES_VPVR, na)

for i_VPVR=0 to MAX_BOXES_VPVR-1

    box.delete(array.get(buyBars_VPVR, i_VPVR))



var sellBars_VPVR = array.new_box(MAX_BOXES_VPVR, na)

for i_VPVR=0 to MAX_BOXES_VPVR-1

    box.delete(array.get(sellBars_VPVR, i_VPVR))



var topLines_VPVR = array.new_line(numOfProfiles_VPVR)

var bottomLines_VPVR = array.new_line(numOfProfiles_VPVR)

var pocLines_VPVR = array.new_line(numOfProfiles_VPVR)

var vahLines_VPVR = array.new_line(numOfProfiles_VPVR)

var valLines_VPVR = array.new_line(numOfProfiles_VPVR)

for i_VPVR=0 to numOfProfiles_VPVR - 1

    line.delete(array.get(topLines_VPVR, i_VPVR))

    line.delete(array.get(bottomLines_VPVR, i_VPVR))

    line.delete(array.get(pocLines_VPVR, i_VPVR))

    line.delete(array.get(vahLines_VPVR, i_VPVR))

    line.delete(array.get(valLines_VPVR, i_VPVR))



var label boxWarning_VPVR = na



if barstate.islast

    if expectedBoxes_VPVR > MAX_BOXES_LIMIT_VPVR

        label.delete(boxWarning_VPVR)

        boxWarning_VPVR := label.new(

            x=bar_index,

            y=high,

            text="⚠ VPVR WARNING\nBoxes requested: " + str.tostring(expectedBoxes_VPVR) +

                 "\nMax allowed: " + str.tostring(MAX_BOXES_LIMIT_VPVR),

            style=label.style_label_down,

            color=color.new(color.red, 0),

            textcolor=color.white,

            size=size.small

        )

    else

        label.delete(boxWarning_VPVR)

   

    for k_VPVR = 0 to numOfProfiles_VPVR - 1

        offset_VPVR = k_VPVR * (numOfBars_VPVR - 1)

        lastBar_VPVR = bar_index - distLastCandle_VPVR - offset_VPVR

        firstBar_VPVR = lastBar_VPVR - numOfBars_VPVR + 1

        visibleBars_VPVR = lastBar_VPVR - firstBar_VPVR + 1

        widestHistWidth_VPVR = visibleBars_VPVR



        histLowList_VPVR = array.new_float(numOfHists_VPVR, na)

        histHighList_VPVR = array.new_float(numOfHists_VPVR, na)

        histPriceList_VPVR = array.new_float(numOfHists_VPVR, 0.0)



        histBuyVolumeList_VPVR = array.new_float(numOfHists_VPVR, 0.0)

        histSellVolumeList_VPVR = array.new_float(numOfHists_VPVR, 0.0)

        histVolumePercentageList_VPVR = array.new_float(numOfHists_VPVR, 0.0)



        totalVolume_VPVR = 0.0

        volumeAccum_VPVR = 0.0

        float VAH_VPVR = na

        float VAL_VPVR = na

        float rangeHigh_VPVR = na

        float rangeLow_VPVR = na

        float rangeHeight_VPVR = na

        float histHeight_VPVR = na



        // Define range height

        rangeHigh_VPVR := high[offset_VPVR + distLastCandle_VPVR]

        rangeLow_VPVR  := low[offset_VPVR + distLastCandle_VPVR]

        for _i_VPVR = 1 to numOfBars_VPVR - 1

            i_VPVR = offset_VPVR + distLastCandle_VPVR + _i_VPVR

            rangeHigh_VPVR := math.max(rangeHigh_VPVR, high[i_VPVR])

            rangeLow_VPVR  := math.min(rangeLow_VPVR, low[i_VPVR])

        rangeHeight_VPVR := rangeHigh_VPVR - rangeLow_VPVR

        histHeight_VPVR := rangeHeight_VPVR / numOfHists_VPVR



        // Draw top and bottom of the range considered

        array.set(topLines_VPVR, k_VPVR, line.new(firstBar_VPVR, rangeHigh_VPVR, lastBar_VPVR, rangeHigh_VPVR, xloc=xloc.bar_index, color=topBottomLineColor_VPVR, width = 5))

        array.set(bottomLines_VPVR, k_VPVR, line.new(firstBar_VPVR, rangeLow_VPVR, lastBar_VPVR, rangeLow_VPVR, xloc=xloc.bar_index, color=topBottomLineColor_VPVR, width = 6))

       

        // Define lows and highs of the hists

        for i_VPVR = 0 to numOfHists_VPVR - 1

            histLow_VPVR = rangeLow_VPVR + histHeight_VPVR * i_VPVR

            histHigh_VPVR = rangeLow_VPVR + histHeight_VPVR * (i_VPVR + 1)

            array.set(histLowList_VPVR, i_VPVR, histLow_VPVR)

            array.set(histHighList_VPVR, i_VPVR, histHigh_VPVR)

            array.set(histPriceList_VPVR, i_VPVR, (histLow_VPVR + histHigh_VPVR) / 2)

           

        // Assign bar's volumes to hists

        for _i_VPVR = 0 to numOfBars_VPVR - 1

            i_VPVR = offset_VPVR + distLastCandle_VPVR + _i_VPVR

            totalVolume_VPVR := totalVolume_VPVR + volume[i_VPVR]

            currentBarHeight_VPVR = high[i_VPVR] - low[i_VPVR]

            currentBuyVolume_VPVR = (high[i_VPVR] == low[i_VPVR]) ? 0 : volume[i_VPVR] * (close[i_VPVR] - low[i_VPVR]) / currentBarHeight_VPVR

            currentSellVolume_VPVR = (high[i_VPVR] == low[i_VPVR]) ? 0 : volume[i_VPVR] * (high[i_VPVR] - close[i_VPVR]) / currentBarHeight_VPVR



            // Define the percentages of the current volume to give to hists

            for j_VPVR = 0 to numOfHists_VPVR - 1

                histLow_VPVR = array.get(histLowList_VPVR, j_VPVR)

                histHigh_VPVR = array.get(histHighList_VPVR, j_VPVR)

                target_VPVR = (math.max(histHigh_VPVR, high[i_VPVR]) - math.min(histLow_VPVR, low[i_VPVR]))

                 - (math.max(histHigh_VPVR, high[i_VPVR]) - math.min(histHigh_VPVR, high[i_VPVR]))

                 - (math.max(histLow_VPVR, low[i_VPVR]) - math.min(histLow_VPVR, low[i_VPVR]))

                histVolumePercentage_VPVR = target_VPVR / currentBarHeight_VPVR

               

                histBuyVolume_VPVR = array.get(histBuyVolumeList_VPVR, j_VPVR)

                histSellVolume_VPVR = array.get(histSellVolumeList_VPVR, j_VPVR)

               

                // If there is at least one hist affected

                // then divide the current volume by the number of hists affected

                if histVolumePercentage_VPVR > 0

                    array.set(histBuyVolumeList_VPVR, j_VPVR, histBuyVolume_VPVR + currentBuyVolume_VPVR * histVolumePercentage_VPVR)

                    array.set(histSellVolumeList_VPVR, j_VPVR, histSellVolume_VPVR + currentSellVolume_VPVR * histVolumePercentage_VPVR)

       

        // Find the hist with the highest volume

        highestHistVolume_VPVR = 0.0

        pocIndex_VPVR = 0

        for i_VPVR = 0 to numOfHists_VPVR - 1

            histBuyVolume_VPVR = array.get(histBuyVolumeList_VPVR, i_VPVR)

            histSellVolume_VPVR = array.get(histSellVolumeList_VPVR, i_VPVR)

            histVolume_VPVR = histBuyVolume_VPVR + histSellVolume_VPVR

            if histVolume_VPVR > highestHistVolume_VPVR

                highestHistVolume_VPVR := histVolume_VPVR

                pocIndex_VPVR := i_VPVR



        // Draw Point of Control

        pocHistLow_VPVR = array.get(histLowList_VPVR, pocIndex_VPVR)

        pocHistHigh_VPVR = array.get(histHighList_VPVR, pocIndex_VPVR)

        pocPos_VPVR = pocHistLow_VPVR + (pocHistHigh_VPVR - pocHistLow_VPVR) / 2

        array.set(pocLines_VPVR, k_VPVR, line.new(firstBar_VPVR, pocPos_VPVR, lastBar_VPVR, pocPos_VPVR, xloc=xloc.bar_index, color=pocLineColor_VPVR, width = 20))  

   

        // Value area calculation

        valueAreaVolume_VPVR  = totalVolume_VPVR * 0.7

        volumeAccum_VPVR := array.get(histBuyVolumeList_VPVR, pocIndex_VPVR) + array.get(histSellVolumeList_VPVR, pocIndex_VPVR)

        VAH_VPVR := pocPos_VPVR

        VAL_VPVR := pocPos_VPVR



        upIndex_VPVR = pocIndex_VPVR + 1

        downIndex_VPVR = pocIndex_VPVR - 1

        for __VPVR = 0 to numOfHists_VPVR - 1

            volUp_VPVR = (upIndex_VPVR < numOfHists_VPVR) ? (array.get(histBuyVolumeList_VPVR, upIndex_VPVR) + array.get(histSellVolumeList_VPVR, upIndex_VPVR)) : 0.0

            volDown_VPVR = (downIndex_VPVR >= 0) ? (array.get(histBuyVolumeList_VPVR, downIndex_VPVR) + array.get(histSellVolumeList_VPVR, downIndex_VPVR)) : 0.0



            if volumeAccum_VPVR >= valueAreaVolume_VPVR

                break



            canGoUp_VPVR = upIndex_VPVR < numOfHists_VPVR

            canGoDown_VPVR = downIndex_VPVR >= 0



            if not canGoUp_VPVR and not canGoDown_VPVR

                break



            if canGoUp_VPVR and (not canGoDown_VPVR or volUp_VPVR >= volDown_VPVR)

                VAH_VPVR := array.get(histPriceList_VPVR, upIndex_VPVR)

                volumeAccum_VPVR += volUp_VPVR

                upIndex_VPVR += 1

            else

                VAL_VPVR := array.get(histPriceList_VPVR, downIndex_VPVR)

                volumeAccum_VPVR += volDown_VPVR

                downIndex_VPVR -= 1



        array.set(vahLines_VPVR, k_VPVR, line.new(firstBar_VPVR, VAH_VPVR, lastBar_VPVR, VAH_VPVR, color=valueAreaLineColor_VPVR, width=8, style=line.style_solid))

        array.set(valLines_VPVR, k_VPVR, line.new(firstBar_VPVR, VAL_VPVR, lastBar_VPVR, VAL_VPVR, color=valueAreaLineColor_VPVR, width=8, style=line.style_solid))



        // Draw hists and highlight the Point of Control

        for i_VPVR = 0 to numOfHists_VPVR - 1

            histLow_VPVR = array.get(histLowList_VPVR, i_VPVR)

            histHigh_VPVR = array.get(histHighList_VPVR, i_VPVR)

            histBuyVolume_VPVR = array.get(histBuyVolumeList_VPVR, i_VPVR)

            histSellVolume_VPVR = array.get(histSellVolumeList_VPVR, i_VPVR)

            histVolume_VPVR = histBuyVolume_VPVR + histSellVolume_VPVR



            histWidth_VPVR = 0.0

            if highestHistVolume_VPVR > 0

                histWidth_VPVR := histVolume_VPVR / highestHistVolume_VPVR * widestHistWidth_VPVR

           

            histBuyWidth_VPVR = 0.0

            histSellWidth_VPVR = 0.0

            if histVolume_VPVR > 0

                histBuyWidth_VPVR := histBuyVolume_VPVR / histVolume_VPVR * histWidth_VPVR

                histSellWidth_VPVR := histSellVolume_VPVR / histVolume_VPVR * histWidth_VPVR



            // Draw buy and sell hists

            boxIndex_VPVR = k_VPVR * numOfHists_VPVR + i_VPVR

            buyHistEnd_VPVR = int(firstBar_VPVR + histBuyWidth_VPVR)

            sellHistEnd_VPVR = int(buyHistEnd_VPVR + histSellWidth_VPVR)

            buyBox_VPVR = box.new(left=firstBar_VPVR, top=histHigh_VPVR, right=buyHistEnd_VPVR, bottom=histLow_VPVR, bgcolor=buyBarColor_VPVR, border_width=0)

            array.set(buyBars_VPVR, boxIndex_VPVR, buyBox_VPVR)

            sellBox_VPVR = box.new(left=buyHistEnd_VPVR, top=histHigh_VPVR, right=sellHistEnd_VPVR, bottom=histLow_VPVR, bgcolor=sellBarColor_VPVR, border_width=0)

            array.set(sellBars_VPVR, boxIndex_VPVR, sellBox_VPVR)
